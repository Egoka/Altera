// Реальная сеть без токенов: TLS до провайдера, отказы host/произвольным адресам.
import http from 'node:http';
import net from 'node:net';
import tls from 'node:tls';

function connect(destination, requestTls = false) {
  return new Promise((resolve, reject) => {
    const request = http.request({ host: 'egress-proxy', port: 8080,
      method: 'CONNECT', path: destination, timeout: 10000 });
    request.on('timeout', () => request.destroy(new Error('timeout')));
    request.on('error', reject);
    request.on('connect', (response, socket) => {
      if (!requestTls || response.statusCode !== 200) {
        socket.destroy(); resolve(response.statusCode); return;
      }
      const hostname = destination.split(':')[0];
      const secure = tls.connect({ socket, servername: hostname }, () => {
        secure.write(`GET / HTTP/1.1\r\nHost: ${hostname}\r\nConnection: close\r\n\r\n`);
      });
      secure.setTimeout(10000, () => secure.destroy(new Error('timeout')));
      secure.on('error', reject);
      secure.once('data', (data) => {
        const status = /^HTTP\/1\.1 (\d+)/.exec(data.toString());
        secure.destroy();
        if (status) resolve(Number(status[1])); else reject(new Error('invalid_http'));
      });
    });
    request.end();
  });
}

function direct(host, port) {
  return new Promise((resolve) => {
    const socket = net.connect({ host, port });
    socket.setTimeout(1500, () => { socket.destroy(); resolve('blocked'); });
    socket.on('connect', () => { socket.destroy(); resolve('CONNECTED'); });
    socket.on('error', () => resolve('blocked'));
  });
}

const forbidden = await Promise.all([
  'host.docker.internal:443', '127.0.0.1:443', '169.254.169.254:443',
  'example.com:443', 'api.openai.com:80',
].map(connect));
const directResults = await Promise.all([
  direct('1.1.1.1', 443), direct('host.docker.internal', 443),
  direct('host.docker.internal', 2375),
]);
const providerTls = await Promise.all([
  connect('api.openai.com:443', true), connect('api.anthropic.com:443', true),
]);
const report = { forbidden, directResults, providerTls, tokenAbsent: !process.env.MULTICA_TOKEN };
process.stdout.write(JSON.stringify(report) + '\n');
process.exit(forbidden.every((code) => code === 403) && directResults.every((value) => value === 'blocked') &&
  providerTls.every((code) => code >= 100 && code <= 599) && report.tokenAbsent ? 0 : 41);
